CALL condenser_si(R$, T_cai, m_dot_r, T_cond, DELTAT_sub : P_cond, h_2, h_3, Q_cond, T_cao)

"! Parametric studies"
$IFNOT PARAMETRIC TABLE

DELTAE = Q_evap + (W_dot - Q_loss) - Q_cond 	"must be close to 0"
